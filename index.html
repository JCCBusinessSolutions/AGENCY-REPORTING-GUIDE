INSTALLATION GUIDE

Agency Reporting App
Setup Guide

Set up the backend of the reporting suite on your own Google Sheet in about 30 minutes: Sheet, Apps Script and web app. No coding needed. Copy, paste, click.

1

What you need before you start

  • A Google account (Gmail). The Sheet and the script live here. Use the account that will own the data permanently.
  • Your Dashboard and Admin Settings links, provided by your installer after Step 6.
  • The backend script (Code.gs). It is in the copy box in Step 3, so you do not need a file.
  • Your own details: your name, advisor code, position, and the email you sign in with.
  • A laptop or desktop for the setup (Apps Script is clumsy on phones). Daily use afterwards works on any phone.
TIPKeep one browser tab on this guide and open each Google tool in new tabs. Tick MARK DONE as you go and you can safely close and resume later.
USE ONE GOOGLE ACCOUNT THROUGHOUTIf you are signed in to several Google accounts, open an Incognito window and sign in only to the account that will own the app. Mixed accounts are the number one cause of "Authorization required" errors.

 *
 * The web app serves the two pages (the dashboard and Admin Settings) and answers their requests.
 * Who can use it: a person whose EMAIL is in REGISTERED USERS. They sign in with that email and a password they create
 * the first time (a 6-digit code is emailed to them to prove the email is theirs). Passwords are stored only as salted hashes in the hidden AUTH_USERS tab.
 * What each person sees:
 *   - Admins (position marked ADMIN in ACCESS_ROLES) see the whole branch.
 *   - Unit managers see production for their unit.
 *   - Everyone else sees only their own production (matched by CODE).
 * Only admins can upload files, change the access list, or change rules.
 *
 * Deploy: Extensions > Apps Script (from this sheet). Create HTML files named
 *   gst-branch-dashboard and gst-rules-settings (paste the two pages into them).

 *
 * Uploads (how it works):
 *   begin  -> clears the staging tabs for this upload

function handle_(body) {

  ensureCleanup_();

  }
  finally { try { lock.releaseLock(); } catch (e2) {} }
}

// Compressed copy of a result in the script cache, split into pieces under the 100 KB limit
function cachePut_(key, obj, ttl) {
  try {
    var gz = Utilities.gzip(Utilities.newBlob(JSON.stringify(obj), "application/json"));
    var b64 = Utilities.base64Encode(gz.getBytes());
    var map = {}, n = 0;
    for (var i = 0; i < b64.length; i += 90000) { map[key + ":" + n] = b64.substring(i, i + 90000); n++; }
    map[key + ":n"] = String(n);
    CacheService.getScriptCache().putAll(map, ttl || 900);
  } catch (e) {}
}
function cacheGet_(key) {
  try {
    var c = CacheService.getScriptCache();
    var n = Number(c.get(key + ":n"));
    if (!n) return null;
    var keys = []; for (var i = 0; i < n; i++) keys.push(key + ":" + i);
    var got = c.getAll(keys), b64 = "";
    for (var j = 0; j < n; j++) { if (got[keys[j]] === undefined || got[keys[j]] === null) return null; b64 += got[keys[j]]; }
    var json = Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b64), "application/x-gzip")).getDataAsString();
    return JSON.parse(json);
  } catch (e) { return null; }
}
function userKey_(user) {
  var raw = [user.code, user.position, user.team, user.unit, user.isAdmin ? 1 : 0].join("|");
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw);
  return d.map(function (b) { return ("0" + (b & 255).toString(16)).slice(-2); }).join("").slice(0, 12);
}

function singlePayDetect_(ss) {
  var m = mdrt_(ss), sp = m.singlePay || {}, min = Number(sp.min) || 250000, rt = Number(sp.roundTo) || 1000;
  var all = prodAll_(ss, "FY_Commission"), hdr = all[0] || [], rows = all.slice(1), pc = -1;

  var by = {}, noPol = 0;
  rows.forEach(function (r) {
    if (/service fee/i.test(String(r[5]))) return;

    o.lines.push({ code: String(Number(r[1]) || r[1]), name: String(r[2] || "").trim(), date: dateText_(ss, r[4]), prem: prem });
    o.total += prem;

  items.forEach(function (o) { o.lines.sort(function (a, b) { return b.prem - a.prem; }); o.total = Math.round(o.total * 100) / 100; });

function regCols_(sh) {
  var out = { status: -1, photo: -1, appt: -1, spin: -1, um: -1, sm: -1, bm: -1, mentor: -1 };

    if (/^APPOINT/.test(t)) {

  }
  out.perPos = out.um >= 0 || out.sm >= 0 || out.bm >= 0;

}

var AUTH_TAB = "AUTH_USERS";

var SESSION_DAYS = 30;
var MAX_TRIES = 5;

function hex_(bytes) {
  return bytes.map(function (b) { return ("0" + (b & 255).toString(16)).slice(-2); }).join("");
}
function sha_(text) {
  return hex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8));
}
function hashPw_(salt, pw) {
  var h = salt + "|" + pw;
  for (var i = 0; i < 50; i++) h = sha_(h + "|" + salt);
  return h;
}
function secret_() {
  var props = PropertiesService.getScriptProperties();
  var s = props.getProperty("GST_SECRET");
  if (!s) { s = Utilities.getUuid() + Utilities.getUuid(); props.setProperty("GST_SECRET", s); }
  return s;
}
function b64_(bytes) { return Utilities.base64EncodeWebSafe(bytes); }
function makeToken_(email) {
  var payload = b64_(Utilities.newBlob(email + "|" + (Date.now() + SESSION_DAYS * 86400000)).getBytes());
  var sig = b64_(Utilities.computeHmacSha256Signature(payload, secret_()));
  return payload + "." + sig;
}
// Returns the email inside a valid, unexpired token, or "" if the token is not valid
function tokenEmail_(token) {
  var parts = String(token || "").split(".");
  if (parts.length !== 2) return "";
  var good = b64_(Utilities.computeHmacSha256Signature(parts[0], secret_()));
  if (good !== parts[1]) return "";
  var text = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString();
  var i = text.lastIndexOf("|");
  if (i < 0 || Number(text.slice(i + 1)) < Date.now()) return "";
  return text.slice(0, i);
}

function signIn_(ss, email, password) {
  email = normEmail_(email);
  password = String(password || "");
  if (!email || !password) return { ok: false, error: "sign_in_required" };
  var cache = CacheService.getScriptCache();
  var failKey = "fail:" + email;
  if (Number(cache.get(failKey) || 0) >= MAX_TRIES) return { ok: false, error: "too_many" };
  var reg = registeredUser_(ss, email);
  if (!reg) return { ok: false, error: "not_registered" };
  var rec = authRow_(ss, email);
  if (!rec) return { ok: false, error: "no_password" };
  if (hashPw_(rec.salt, password) !== rec.hash) {
    cache.put(failKey, String(Number(cache.get(failKey) || 0) + 1), 900);
    return { ok: false, error: "bad_password" };
  }
  cache.remove(failKey);
  logLogin_(ss, reg);
  var token = makeToken_(email);
  return { ok: true, token: token, user: userWithAccess_(ss, reg), branchName: getSetting_(ss, "BRANCH NAME"), branchLogo: getSetting_(ss, "BRANCH LOGO"), themeColor: getSetting_(ss, "THEME COLOR") };
}

// Leading rows of a log (oldest first) whose date is before the cutoff: deletes them. Returns how many were removed.
function trimOldRows_(sh, cutoff, parse) {
  var n = sh.getLastRow() - 1; if (n < 1) return 0;
  var col = sh.getRange(2, 1, n, 1).getValues(), k = 0;
  while (k < n) { var d = parse(col[k][0]); if (!d || d >= cutoff) break; k++; }
  if (k > 0) sh.deleteRows(2, k);
  return k;
}
function cleanupRun_(ss, haveLock) {
  var lock = null;
  if (!haveLock) { lock = LockService.getScriptLock(); if (!lock.tryLock(10000)) return { ok: false, error: "busy" }; }
  var res = { at: Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone() || "Asia/Manila", "yyyy-MM-dd HH:mm"), logins: 0, uploadLog: 0, apps: 0, receipts: 0, stagingTabs: 0, notes: [] };
  try {
    var now = new Date();
    // an upload in the last 15 minutes may still be using the staging tabs: leave those for the next run
    var uploading = false, ul = ss.getSheetByName(LOG_TAB);
    if (ul && ul.getLastRow() > 1) { var lastUp = ul.getRange(ul.getLastRow(), 1).getValue(); if (lastUp instanceof Date && now - lastUp < 15 * 60000) uploading = true; }
    // 1) sign-ins
    var lg = ss.getSheetByName(LOGIN_TAB);
    if (lg) {
      var c1 = Utilities.formatDate(new Date(now.getTime() - CLEAN_CFG.loginDays * 86400000), ss.getSpreadsheetTimeZone() || "Asia/Manila", "yyyy-MM-dd HH:mm:ss");
      res.logins = trimOldRows_(lg, c1, function (v) { var s = String(v || "").trim(); return /^\d{4}-\d{2}-\d{2}/.test(s) ? s : ""; });
    }
    // 2) upload log
    if (ul) { var c2 = new Date(now.getTime()); c2.setMonth(c2.getMonth() - CLEAN_CFG.uploadLogMonths); res.uploadLog = trimOldRows_(ul, c2, function (v) { return v instanceof Date ? v : null; }); }
    // 3) applications that are no longer PENDING (settled, declined, postponed, NPW) are deleted every quarter:
    //    those submitted before the start of the current quarter go. Pending ones always stay until they are closed.
    var ap = ss.getSheetByName(APPS_TAB);
    if (ap && ap.getLastRow() > 1) {
      var w = APPS_HEADER.length, rows = ap.getRange(2, 1, ap.getLastRow() - 1, w).getValues(), info = {};
      var nd = new Date(), qStart = new Date(nd.getFullYear(), Math.floor(nd.getMonth() / 3) * 3, 1);
      rows.forEach(function (r) {
        var id = String(r[0]); if (!id) return;
        var o = info[id] || (info[id] = { pending: false, old: true });
        var st = String(r[22] || "PENDING").trim().toUpperCase(); if (!CLEAN_CLOSED[st]) o.pending = true;
        var d = r[2] instanceof Date ? r[2] : new Date(String(r[2]).slice(0, 10) + "T12:00:00"); if (!(d instanceof Date) || isNaN(d) || d >= qStart) o.old = false;
      });
      var del = [];
      rows.forEach(function (r, i) { var o = info[String(r[0])]; if (o && o.old && !o.pending) del.push(i + 2); });
      for (var j = del.length - 1; j >= 0; j--) ap.deleteRow(del[j]);
      res.apps = del.length;
    }
    if (!uploading) {
      // 4) leftover chunk receipts
      var rc = ss.getSheetByName(RECEIPTS_TAB);
      if (rc && rc.getLastRow() > 0) {
        var rv = rc.getRange(1, 1, rc.getLastRow(), 4).getValues(), keep = [], limit = now.getTime() - CLEAN_CFG.receiptDays * 86400000;
        rv.forEach(function (r) { var m = String(r[0]).match(/-(\d{10,})$/); if (m && Number(m[1]) < limit) res.receipts++; else keep.push(r); });
        if (res.receipts) { rc.clearContents(); if (keep.length) rc.getRange(1, 1, keep.length, 4).setValues(keep); }
      }
      // 5) staging tabs that are empty but still carry thousands of blank rows
      ss.getSheets().forEach(function (sh) {
        var nm = sh.getName();
        if (nm.indexOf(STAGE_PREFIX) !== 0 || nm === RECEIPTS_TAB) return;
        if (sh.getLastRow() > 0 || sh.getMaxRows() <= 2) return;
        sh.deleteRows(2, sh.getMaxRows() - 1); res.stagingTabs++;
      });
    } else res.notes.push("An upload was running, so staging tabs and receipts were left for the next run.");
  } catch (e) { res.ok = false; res.error = String(e && e.message || e).slice(0, 200); }
  finally { if (lock) lock.releaseLock(); }
  if (res.ok !== false) res.ok = true;
  try { PropertiesService.getScriptProperties().setProperty("CLEANUP_LAST", JSON.stringify(res)); } catch (e) {}
  return { ok: res.ok, last: res, cfg: CLEAN_CFG, error: res.error };
}

var PROD_BASES = ["Settled", "Submitted", "FY_Commission"];
function prodTabName_(base, slot) { var y = new Date().getFullYear(); return (slot === "previous" ? y - 1 : y) + "_" + base; }
// An upload addressed to "Settled" / "Submitted" / "FY_Commission" is written to the tab of its year (slot current = this year, previous = last year)
function prodRemap_(body) {
  var slot = body.slot === "previous" ? "previous" : ((body.slot === "current" || body.slot === "daily") ? "current" : "");   // "daily" adds to this year's tabs
  if (!slot) return body;
  function m(t) { return PROD_BASES.indexOf(t) >= 0 ? prodTabName_(t, slot) : t; }
  if (body.tab) body.tab = m(body.tab);
  (body.tables || []).forEach(function (t) { t.tab = m(t.tab); });
  return body;
}
// One time per year: the old single tabs (Settled, Submitted, FY_Commission) become this year's tab, and last year's tab is created empty
function prodMigrate_(ss) {
  var props = PropertiesService.getScriptProperties(), flag = "PRODTABS_" + new Date().getFullYear();
  if (props.getProperty(flag)) return;
  PROD_BASES.forEach(function (b) {
    var legacy = ss.getSheetByName(b), cur = prodTabName_(b, "current"), prev = prodTabName_(b, "previous");
    var header = legacy && legacy.getLastRow() > 0 ? legacy.getRange(1, 1, 1, legacy.getLastColumn()).getValues()[0] : null;
    if (legacy && !ss.getSheetByName(cur)) legacy.setName(cur);
    if (!ss.getSheetByName(prev)) {
      var sh = ss.insertSheet(prev);
      if (header && header.length) { sh.getRange(1, 1, 1, header.length).setValues([header]).setFontWeight("bold"); sh.setFrozenRows(1); }
    }
  });
  props.setProperty(flag, "1");
}
// All rows of a report (header first) from this year's and last year's tabs. A row that appears in both (the cut-off that spans New Year, or the same file uploaded twice) is counted once.
function prodAll_(ss, base) {
  var keyCols = base === "Settled" ? [1, 4, 5, 6, 7] : base === "Submitted" ? [1, 4, 5, 7, 8] : [1, 4, 5, 6, 7, 8, 9];
  var names = [prodTabName_(base, "current"), prodTabName_(base, "previous")], header = [], out = [], counts = null;
  names.forEach(function (n, ti) {
    var sh = ss.getSheetByName(n);
    if (!sh || sh.getLastRow() < 1 || sh.getLastColumn() < 1) return;
    var vals = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
    if (!header.length) header = vals[0];
    var rows = vals.slice(1).filter(function (r) { return String(r[1]).trim() !== ""; });
    function key(r) { return keyCols.map(function (i) { return r[i] instanceof Date ? cell_(ss, r[i], true) : String(r[i]); }).join("|"); }
    if (ti === 0) { counts = {}; rows.forEach(function (r) { var k = key(r); counts[k] = (counts[k] || 0) + 1; }); out = out.concat(rows); }
    else rows.forEach(function (r) { var k = counts ? key(r) : ""; if (counts && counts[k] > 0) { counts[k]--; return; } out.push(r); });
  });
  return [header].concat(out);
}

// A manager (or manager candidate) adds a candidate from the dashboard (status starts at the first one). Unit manager: own unit only. Sales manager: their units or themselves as direct manager. Branch manager and admin: anyone.

  if (!rv.allowed) return { ok: false, error: "Recruitment is for managers only." };

// ---------- ACTIVATION tab: who the signed-in person may see, and submitted / settled lives per advisor per day ----------
// Branch view (admin, branch manager, ADMIN, SECRETARY): everyone. Sales manager: the sales team. Unit manager: the unit plus the manager candidates under the unit manager
// (and their units), found through MENTOR CODE. Manager candidate: the own unit. Anyone else: themselves.
function activation_(ss, user) {
  var users = users_(ss), me = normCode_(user.code), pos = user.position, vis = {}, kids = {}, byCode = {};
  users.forEach(function (u) {
    var c = normCode_(u[1]); if (!c) return;
    byCode[c] = u;
    var mc = normCode_(u[13]); if (mc && mc !== c) (kids[mc] = kids[mc] || []).push(c);
  });
  var uKey = unitKey_(user.unit), team = String(user.team || "").trim().toUpperCase();
  if (branchView_(user)) users.forEach(function (u) { vis[normCode_(u[1])] = 1; });
  else if (pos === "SALES MANAGER" && team) users.forEach(function (u) { if (String(u[4]).trim().toUpperCase() === team) vis[normCode_(u[1])] = 1; });
  else if (pos === "UNIT MANAGER" || /CANDIDATE/.test(pos)) {
    var units = {}; if (uKey) units[uKey] = 1;
    if (pos === "UNIT MANAGER") {
      var queue = [me], seen = {}; seen[me] = 1;
      while (queue.length) { var c0 = queue.shift(); (kids[c0] || []).forEach(function (k) { if (seen[k]) return; seen[k] = 1; queue.push(k); vis[k] = 1; var uu = byCode[k]; if (uu && /CANDIDATE|MANAGER/.test(String(uu[2]).toUpperCase())) units[unitKey_(uu[5])] = 1; }); }
    }
    users.forEach(function (u) { if (units[unitKey_(u[5])]) vis[normCode_(u[1])] = 1; });
    if (me) vis[me] = 1;
  } else if (me) vis[me] = 1;
  var roster = [];
  users.forEach(function (u) { var c = normCode_(u[1]); if (c && vis[c]) roster.push([String(u[1]).trim(), String(u[0]).trim(), String(u[2]).trim().toUpperCase(), String(u[4]).trim(), String(u[5]).trim(), String(u[9] || "").trim().toUpperCase(), String(u[13] || "").trim(), dateText_(ss, u[7])]); });
  var from = (new Date().getFullYear() - 2) + "-12-01", agg = {}, out = [];
  function take(base, slot) {
    prodAll_(ss, base).slice(1).forEach(function (r) {
      var c = normCode_(r[1]); if (!c || !vis[c]) return;
      var d = String(cell_(ss, r[4], true) || ""); if (!d || d < from) return;
      var k = c + "|" + d, o = agg[k];
      if (!o) { o = agg[k] = [String(r[1]).trim(), d, 0, 0]; out.push(o); }
      o[slot] += Number(r[5]) || 0;
    });
  }
  take("Submitted", 2); take("Settled", 3);
  return { ok: true, roster: roster, rows: out };
}

function qeList_(ss, key) { try { var v = JSON.parse(String(getSetting_(ss, key) || "[]")); return Array.isArray(v) ? v : []; } catch (e) { return []; } }
function quickRaw_(ss) { return { ok: true, links: qeList_(ss, "QUICK LINKS"), events: qeList_(ss, "UPCOMING EVENTS"), cats: qeList_(ss, "QUICK LINK CATEGORIES"), evcats: qeList_(ss, "UPCOMING EVENT CATEGORIES") }; }
function quickEvents_(ss) {

  r.events.sort(function (a, b) { return a.d < b.d ? -1 : a.d > b.d ? 1 : 0; });
  return { links: r.links, events: r.events, cats: r.cats, evcats: r.evcats };

function setQuickEvents_(ss, user, links, events, cats, evcats) {
  if (!user.isAdmin) return { ok: false, error: "admin_only" };
  function cl(v, n) { return String(v == null ? "" : v).replace(/\s+/g, " ").trim().slice(0, n); }
  function url(v, what, n) { v = cl(v, 1000); if (v && !/^https?:\/\//i.test(v)) throw new Error(what + " of row " + n + " must start with https://"); return v; }
  var L = [], E = [];
  try {
    (links || []).forEach(function (x, i) {
      var t = cl(x.t, 80), u = url(x.u, "The link", i + 1), l = url(x.l, "The logo link", i + 1), c = cl(x.c, 40);
      if (!t && !u && !l) return;
      if (!t) throw new Error("Quick link row " + (i + 1) + " needs a name.");
      if (!u) throw new Error("Quick link row " + (i + 1) + " needs a link.");
      L.push({ t: t, u: u, l: l, c: c });
    });
    (events || []).forEach(function (x, i) {
      var t = cl(x.t, 100), d = cl(x.d, 10), u = url(x.u, "The registration link", i + 1), l = url(x.l, "The logo link", i + 1);
      if (!t && !d && !u && !l) return;
      if (!t) throw new Error("Event row " + (i + 1) + " needs a name.");
      if (!/^\d{4}-\d{2}-\d{2}$/.test(d)) throw new Error("Event row " + (i + 1) + " needs a date.");
      E.push({ t: t, d: d, tm: cl(x.tm, 30), p: cl(x.p, 100), u: u, l: l, c: cl(x.c, 40) });
    });
  } catch (e) { return { ok: false, error: e.message }; }
  if (L.length > 40) return { ok: false, error: "Use at most 40 quick links." };
  if (E.length > 100) return { ok: false, error: "Use at most 100 events." };
  var C = [], seen = {};
  (cats || []).forEach(function (c) { c = cl(c, 40); if (c && !seen[c.toLowerCase()]) { seen[c.toLowerCase()] = 1; C.push(c); } });
  var EC = [], seen2 = {};
  (evcats || []).forEach(function (c) { c = cl(c, 40); if (c && !seen2[c.toLowerCase()]) { seen2[c.toLowerCase()] = 1; EC.push(c); } });
  setSetting_(ss, "UPCOMING EVENT CATEGORIES", JSON.stringify(EC));
  setSetting_(ss, "QUICK LINK CATEGORIES", JSON.stringify(C));
  setSetting_(ss, "QUICK LINKS", JSON.stringify(L));
  setSetting_(ss, "UPCOMING EVENTS", JSON.stringify(E));
  return quickRaw_(ss);
}

  function body_(name, cols) {
    var sh = ss.getSheetByName(name);
    if (!sh || sh.getLastRow() < 2) return [];
    return sh.getRange(2, 1, sh.getLastRow() - 1, Math.min(cols, sh.getLastColumn())).getValues();
  }

  [["Settled", 9, 4, 8], ["Submitted", 7, 4, 6], ["FY_Commission", 11, 4, 10]].forEach(function (t) {
    ["current", "previous"].forEach(function (src) {
      body_(prodTabName_(t[0], src), t[1]).forEach(function (r) {
        if (String(r[1]).trim() === "") return;
        var o = prod[src][t[0]] || (prod[src][t[0]] = { rows: 0, min: "", max: "", ac: 0, at: "" });
        o.rows++;
        var d = String(cell_(ss, r[t[2]], true) || "");
        if (d && (!o.min || d < o.min)) o.min = d;
        if (d && d > o.max) o.max = d;
        if (t[0] === "Settled") o.ac += Number(r[6]) || 0;
        var at = r[t[3]] instanceof Date ? r[t[3]].toISOString() : String(r[t[3]] || "");
        if (at > o.at) o.at = at;
      });
    });
  });
  var pers = [];
  ss.getSheets().forEach(function (sh) {
    var m = sh.getName().match(/^(13|25)th_([^_]+)(?:_(Actual|Projected))?(_.*)?$/i);
    if (!m) return;
    pers.push({ tab: sh.getName(), slot: m[1] === "13" ? "p13" : "p25", period: m[2].trim().toUpperCase(), kind: m[3] ? m[3].toLowerCase() : "", extra: !!m[4], rows: Math.max(0, sh.getLastRow()) });
  });
  return { ok: true, production: prod, persistency: pers };
}

function setCutoffs_(ss, user, list) {

  }

  }
  return { ok: true, cutoffs: rows };
}

// 13th and 25th month persistency. One set of tabs per cut-off period:
//   13th_<Month YYYY>_Actual / _Projected and 25th_<Month YYYY>_Actual / _Projected  (for example 13th_October 2026_Projected)
//   Other sheets of the same file are saved as 13th_October 2026_Projected_<sheet>.
//   Actual or Projected comes from the "Type" row of the report itself.
// Older tabs P13_<PERIOD>_<sheet> / P25_<PERIOD>_<sheet> and P13_<sheet> (no period) are still read.
function persTabs_(ss) {
  var list = [];
  ss.getSheets().forEach(function (sh) {
    var n = sh.getName();
    if (/agreement/i.test(n)) return;
    // Tab names can be cut short by the sheet (for example "P13_SEPTEMBER 2026_Manager Pers"), so only the period part is relied on
    var m = n.match(/^(13|25)th_([A-Za-z]+ \d{4})(?:_(Actual|Projected))?(?:_(.*))?$/i);
    if (m) { list.push({ slot: "p" + m[1], period: m[2].trim().toUpperCase(), sh: sh, main: !m[4], kind: (m[3] || "").toLowerCase() }); return; }
    m = n.match(/^P(13|25)_([^_]+)_(.*)$/);
    if (m) { list.push({ slot: "p" + m[1], period: m[2].trim().toUpperCase(), sh: sh, main: false, kind: "" }); return; }
    m = n.match(/^P(13|25)_(.+)$/);
    if (m && /pers/i.test(m[2])) list.push({ slot: "p" + m[1], period: "", sh: sh, main: false, kind: "" });
  });
  return list;
}
function readPers_(sh, user, maps, onlyCodes) {
  var res = { rows: [], totals: [], cutoff: "", measure: "", kind: "", type: "" };
  if (!sh || sh.getLastRow() < 1 || sh.getLastColumn() < 1) return res;
  var all = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
  var hi = -1, mgrCode = "", mgrTotal = 0, mgrName = "";
  all.forEach(function (r, i) {
    if (String(r[0]) === "Cutoff Period") res.cutoff = String(r[1]);
    if (String(r[0]).trim() === "Type" && !res.kind) {
      var tt = String(r[1]).trim();
      if (/^projected/i.test(tt)) { res.kind = "projected"; res.type = tt; }
      else if (/^actual/i.test(tt)) { res.kind = "actual"; res.type = tt; }
    }
    if (String(r[0]) === "Business Measurement") res.measure = String(r[1]).replace(/\s+/g, " ");
    if (hi < 0 && String(r[0]) === "Manager Code" && String(r[2]) === "Manager Class Code") hi = i;
    if (!mgrCode && String(r[0]) === "Manager Code" && Number(r[1]) > 0) mgrCode = String(Number(r[1]));
  });
  // Columns: E advisor code, F name, G status, D unit, P combined premium ratio (the persistency %)
  if (hi >= 0) {
    for (var i = hi + 1; i < all.length; i++) {
      var r = all[i];
      if (!(Number(r[4]) > 0)) {
        // A unit total row (no advisor code): keep the labels and the combined ratio, the dashboard matches the unit code
        var tr = Number(r[15]);
        // The first "<Manager> Totals" row (no unit in brackets) is the owner's own figure: the whole team's persistency
        if (!mgrTotal && tr > 0 && /Totals\s*$/i.test(String(r[0]).trim()) && String(r[0]).indexOf("(") < 0) mgrTotal = tr;
        if (tr > 0 && (String(r[0]).trim() || String(r[1]).trim() || String(r[2]).trim() || String(r[3]).trim())) res.totals.push([String(r[0]).trim(), String(r[1]).trim(), String(r[2]).trim(), String(r[3]).trim(), tr, String(r[5]).trim()]);
        continue;
      }
      if (onlyCodes) { if (!onlyCodes[normCode_(r[4])]) continue; }
      else if (!canSee_(user, r[4], r[3], maps)) continue;
      res.rows.push([String(Number(r[4])), String(r[5]), String(r[6]), Number(r[15]) || 0, String(r[3])]);
    }
  }
  // The report is run for its owner (Manager Code): the dashboard uses that code to pick the top Totals row for the owner's validation, while the personal row stays the personal figure
  if (mgrCode) res.mgrCode = mgrCode;
  return res;
}
// The latest uploaded report of each kind (the one with the latest cut-off period)
function persistency_(ss, user, maps, onlyCodes) {
  var endOf = {};
  cutoffs_(ss).forEach(function (c) { endOf[String(c.period).toUpperCase()] = c.end; });
  var out = { p13: { rows: [], totals: [], cutoff: "", measure: "", period: "" }, p25: { rows: [], totals: [], cutoff: "", measure: "", period: "" } };
  ["p13", "p25"].forEach(function (slot) {
    var best = null, bestKey = "";
    persTabs_(ss).forEach(function (t) {
      if (t.slot !== slot) return;
      var key = t.period ? (endOf[t.period] || "0000-00-00") : "";
      if (!best || key > bestKey || (key === bestKey && ((t.kind === "actual" && best.kind !== "actual") || (t.kind === best.kind && t.main && !best.main)))) { best = t; bestKey = key; }
    });
    if (best) { var res = readPers_(best.sh, user, maps, onlyCodes); res.period = best.period; out[slot] = res; }
  });
  return out;
}
// Every uploaded period for the people on the roster: { "OCTOBER 2026": { p13: {rows}, p25: {rows} } }
function persPeriods_(ss, codes) {
  var out = {};
  persTabs_(ss).forEach(function (t) {
    if (!t.period) return;
    if (!out[t.period]) out[t.period] = {};
    var got = readPers_(t.sh, null, null, codes), had = out[t.period][t.slot];
    // An Actual report always wins over a Projected one for the same month
    if (had && had.kind === "actual" && got.kind !== "actual") return;
    if (!had || (got.kind === "actual" && had.kind !== "actual") || (!had.rows.length && got.rows.length) || (t.main && got.rows.length)) out[t.period][t.slot] = got;
  });
  return out;
}


// ---------- EAMB (mentor bonus): % of the AC of a manager's Direct, Indirect 1 and Indirect 2 advisors ----------
// Who is whose mentor comes from MENTOR CODE in the masterlist (Direct = mentor is me; Indirect 1 = mentor is one of my Direct; Indirect 2 = mentor is one of my Indirect 1).
// Same cut-off periods as Plan B: AC of the advisor in the period, NSC of the period and the two before it picks the band.
// Rates are % of AC. Columns follow the NSC bands below; edit here when Sun Life changes them.
// Switched off for now: an advisor who needs persistency but is in neither the 13th nor the 25th month report is NOT counted (set true to treat the advisor as under 16 months instead)
var ASSUME_UNDER_16_IF_MISSING = false;
var EAMB_BANDS = [50000, 100000, 200000, 300000];
var EAMB_RATES = {
  NEW:    { direct: [35, 40, 45, 50], ind1: [10, 14, 18, 24], ind2: [5, 6, 8, 12] },     // New agents: 3 years (36 months) or less
  TENURED: { direct: [6, 9, 15, 25], ind1: [2, 3, 6, 12], ind2: [1, 1.5, 3, 6] }        // Tenured agents: more than 3 years
};
function monthsBetween_(fromIso, toIso) {
  var m = String(fromIso || "").match(/^(\d{4})-(\d{2})-(\d{2})$/), n = String(toIso || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!m || !n) return null;
  var mo = (+n[1] - +m[1]) * 12 + (+n[2] - +m[2]);
  if (+n[3] < +m[3]) mo--;
  return Math.max(0, mo);
}
function eamb_(ss, user) {
  var me = normCode_(user.code);
  if (!me) return null;
  var users = users_(ss), kids = {}, info = {};
  users.forEach(function (u) {
    var c = normCode_(u[1]); if (!c) return;
    info[c] = { name: String(u[0]), unit: String(u[5]), coding: String(u[7] || "") };
    var mc = normCode_(u[13]); if (!mc || mc === c) return;
    (kids[mc] = kids[mc] || []).push(c);
  });
  var level = {}, l1 = kids[me] || [], l2 = [], l3 = [];
  l1.forEach(function (c) { level[c] = "direct"; });
  l1.forEach(function (c) { (kids[c] || []).forEach(function (d) { if (!level[d] && d !== me) { level[d] = "ind1"; l2.push(d); } }); });
  l2.forEach(function (c) { (kids[c] || []).forEach(function (d) { if (!level[d] && d !== me) { level[d] = "ind2"; l3.push(d); } }); });
  var codes = Object.keys(level);
  if (!codes.length) return { agents: 0, months: {} };
  var cuts = cutoffs_(ss), thisYear = String(new Date().getFullYear()), today = Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), "yyyy-MM-dd");
  // Settled production of the downline: code, date, AC, NSC
  var sRows = prodAll_(ss, "Settled").slice(1), prod = {};
  if (sRows.length) {
    sRows.forEach(function (r) {
      var c = normCode_(r[1]); if (!level[c]) return;
      var d = cell_(ss, r[4], true); if (!d) return;
      (prod[c] = prod[c] || []).push([String(d), Number(r[6]) || 0, Number(r[7]) || 0]);
    });
  }
  var rc = {}; codes.forEach(function (c) { rc[c] = 1; });
  var perP = persPeriods_(ss, rc), latest = persistency_(ss, null, null, rc);
  function pmap(res) { var m = {}; ((res && res.rows) || []).forEach(function (x) { var k = normCode_(x[0]); if (!(k in m)) m[k] = Number(x[3]) || 0; }); return m; }
  var out = { agents: codes.length, months: {} };
  cuts.forEach(function (cur, i) {
    if (i < 2 || cur.end.slice(0, 4) < thisYear || cur.start > today) return;
    var c3 = cuts[i - 2], key = String(cur.period).toUpperCase(), isNow = cur.start <= today && today <= cur.end;
    var asOf = cur.end < today ? cur.end : today;
    var pp = perP[key] || null;
    var m13 = pp ? pmap(pp.p13) : (isNow ? pmap(latest.p13) : {}), m25 = pp ? pmap(pp.p25) : (isNow ? pmap(latest.p25) : {});
    var hasP = !!pp || (isNow && (Object.keys(m13).length > 0 || Object.keys(m25).length > 0));
    var res = { direct: { ac: 0, eamb: 0, n: 0 }, ind1: { ac: 0, eamb: 0, n: 0 }, ind2: { ac: 0, eamb: 0, n: 0 }, total: 0, pending: 0, list: [] };
    codes.forEach(function (c) {
      var ac = 0, nsc3 = 0;
      (prod[c] || []).forEach(function (x) {
        if (x[0] >= cur.start && x[0] <= cur.end) ac += x[1];
        if (x[0] >= c3.start && x[0] <= cur.end) nsc3 += x[2];
      });
      ac = r2_(ac); nsc3 = r2_(nsc3);
      if (!(ac > 0) && !(nsc3 > 0)) return;
      var lv = level[c], a = info[c] || { name: c, unit: "", coding: "" }, ten = monthsBetween_(a.coding, asOf);
      var tenured = ten != null && ten > 36, status = "ok", pct = 0, eamb = 0, assumed = 0;
      if (ten == null) status = "no_coding";
      else {
        var p13 = (c in m13) ? m13[c] : null, p25 = (c in m25) ? m25[c] : null;
        // Same tenure rule as Plan B: under 16 months no persistency; 16 to 27 months the 13th month (86%); 28 months and up the 25th month (80%)
        var which = ten < 16 ? null : (ten < 28 ? "p13" : "p25");
        // Tenure decides which persistency counts (no fallback to the other one). Reports uploaded but the advisor is not in the report that is needed: persistency counts as 0%
        if (which && hasP && (which === "p13" ? p13 : p25) == null) { assumed = 2; if (which === "p13") p13 = 0; else p25 = 0; }
        if (which === "p13") {
          if (p13 == null) status = "no_persistency"; else if (p13 < 86) status = "below";
        } else if (which === "p25") {
          if (p25 == null) status = "no_persistency"; else if (p25 < 80) status = "below";
        }
      }
      if (status === "ok") {
        var bi = -1; EAMB_BANDS.forEach(function (b, k) { if (nsc3 >= b) bi = k; });
        if (bi < 0) status = "low_nsc";
        else { pct = EAMB_RATES[tenured ? "TENURED" : "NEW"][lv][bi]; eamb = r2_(ac * pct / 100); }
      }
      var L = res[lv]; L.ac = r2_(L.ac + ac); L.eamb = r2_(L.eamb + eamb); L.n++;
      res.total = r2_(res.total + eamb);
      if (status === "no_persistency" || status === "no_coding") res.pending++;
      res.list.push([a.name, a.unit, lv, nsc3, ac, pct, eamb, status, tenured ? "T" : "N", ten, (c in m13) ? m13[c] : null, (c in m25) ? m25[c] : null, pp ? 1 : 0, (c in prod) ? 1 : 0, assumed]);
    });
    out.months[key] = res;
  });
  return out;
}

  users.forEach(function (u) {
    var c = normCode_(u[1]); if (!c) return;
    pos[c] = String(u[2] || "").toUpperCase(); nm[c] = String(u[0]);
    var mc = normCode_(u[13]); if (!mc || mc === c) return;
    (kids[mc] = kids[mc] || []).push(c);
  });
  var group = {}, directs = kids[me] || [];
  directs.forEach(function (c) { group[c] = "direct"; });
  directs.forEach(function (d) {
    var kind = /CANDIDATE/.test(pos[d] || "") ? "mc" : (/UNIT MANAGER/.test(pos[d] || "") ? "um" : null);
    if (!kind) return;
    owner[d] = d;
    var stack = (kids[d] || []).slice(), guard = 0;
    while (stack.length && guard++ < 5000) {
      var c = stack.pop();
      if (group[c] || c === me) continue;
      group[c] = kind; owner[c] = d;
      (kids[c] || []).forEach(function (x) { stack.push(x); });
    }
  });
  var codes = Object.keys(group);
  if (!codes.length) return { agents: 0, months: {} };
  var cl = closings_(ss), thisYear = String(new Date().getFullYear());
  cl = cl.filter(function (c) { return c.me.slice(0, 4) === thisYear; });
  if (!cl.length) return { agents: codes.length, months: {} };
  var sRows = prodAll_(ss, "Settled").slice(1), prod = [];
  if (sRows.length) {
    sRows.forEach(function (r) {
      var g = group[normCode_(r[1])]; if (!g) return;
      var d = cell_(ss, r[4], true); if (!d) return;
      prod.push([String(d), g, Number(r[6]) || 0, owner[normCode_(r[1])]]);
    });
  }
  var yStart = thisYear + "-01-01", out = { agents: codes.length, counts: { direct: 0, um: 0, mc: 0 }, months: {} };
  codes.forEach(function (c) { out.counts[group[c]]++; });
  cl.forEach(function (m, i) {
    var start = i > 0 ? isoAddDays_(cl[i - 1].me, 1) : yStart;
    var res = { total: 0 };
    var calc = function (k, own) {
      var ytd = 0, ac = 0;
      prod.forEach(function (x) {
        if (x[1] !== k || x[0] < yStart || x[0] > m.me || (own && x[3] !== own)) return;
        ytd += x[2];
        if (x[0] >= start) ac += x[2];
      });
      ytd = r2_(ytd); ac = r2_(ac);
      var rate = orcRate_(k, ytd);
      return { ytd: ytd, ac: ac, rate: rate, orc: r2_(ac * rate / 100) };
    };
    res.direct = calc("direct"); res.direct.n = out.counts.direct; res.total = res.direct.orc;
    ["um", "mc"].forEach(function (k) {
      var items = [];
      directs.forEach(function (d) {
        if (!owner[d] || group[d] !== "direct" || (k === "mc") !== /CANDIDATE/.test(pos[d] || "")) return;
        var kind = /CANDIDATE/.test(pos[d] || "") ? "mc" : "um"; if (kind !== k) return;
        var g = calc(k, d); g.name = nm[d] || d;
        g.n = Object.keys(owner).filter(function (c) { return owner[c] === d && c !== d; }).length;
        items.push(g);
        res.total = r2_(res.total + g.orc);
      });
      res[k] = { items: items, orc: r2_(items.reduce(function (t, g) { return t + g.orc; }, 0)) };
    });
    out.months[String(m.month).toUpperCase()] = res;
  });
  return out;
}

// Unit Manager (12%) and Manager Candidate (8%): a flat rate on the AC settled in the month by everyone under the person (mentor chain, any depth), not the person's own AC
function orcFlat_(ss, me, rate, posU) {
  var users = users_(ss), kids = {};
  users.forEach(function (u) {
    var c = normCode_(u[1]); if (!c) return;
    var mc = normCode_(u[13]); if (!mc || mc === c) return;
    (kids[mc] = kids[mc] || []).push(c);
  });
  var under = {}, stack = (kids[me] || []).slice(), guard = 0;
  while (stack.length && guard++ < 5000) {
    var c = stack.pop();
    if (under[c] || c === me) continue;
    under[c] = 1;
    (kids[c] || []).forEach(function (x) { stack.push(x); });
  }
  var n = Object.keys(under).length;
  var cl = closings_(ss), thisYear = String(new Date().getFullYear());
  cl = cl.filter(function (c) { return c.me.slice(0, 4) === thisYear; });
  var out = { agents: n, flat: true, months: {} };
  if (!n || !cl.length) return out;
  var sRows = prodAll_(ss, "Settled").slice(1), prod = [];
  if (sRows.length) {
    sRows.forEach(function (r) {
      if (!under[normCode_(r[1])]) return;
      var d = cell_(ss, r[4], true); if (!d) return;
      prod.push([String(d), Number(r[6]) || 0]);
    });
  }
  var label = /CANDIDATE/.test(posU) ? "Manager Candidate ORC" : "Unit Manager ORC";
  cl.forEach(function (m, i) {
    var start = i > 0 ? isoAddDays_(cl[i - 1].me, 1) : thisYear + "-01-01", ac = 0;
    prod.forEach(function (x) { if (x[0] >= start && x[0] <= m.me) ac += x[1]; });
    ac = r2_(ac);
    var orc = r2_(ac * rate / 100);
    out.months[String(m.month).toUpperCase()] = { total: orc, flat: { label: label, rate: rate, ac: ac, n: n } };
  });
  return out;
}

function growthAdv_(a, dflt) {
  var d = dflt || defaultGrowthAdv_(); if (!a || typeof a !== "object") return d;

  }).sort(function (x, y) { return x.min - y.min; });
  if (!bands.length) return d;
  bands[0].min = 0;
  return { month: /^\d{4}-\d{2}$/.test(String(a.month || "")) ? a.month : d.month, minManpower: Math.round(num(a.minManpower)), asOf: /^\d{4}-\d{2}-\d{2}$/.test(String(a.asOf || "")) ? a.asOf : "", cols: cols, bands: bands };

  if (c) {

// Serves one Google Drive MP3 from the music list as base64, because browsers (iPhone especially) often refuse to stream Drive links directly.
// Only files that are in the saved music list can be fetched (admins can also test a file before saving).
function musicFile_(ss, user, id) {
  id = String(id || "").trim();
  if (!/^[\w-]{10,}$/.test(id)) return { ok: false, error: "That Drive file link is not valid." };
  if (!user.isAdmin) {
    var listed = awardsMusic_(ss).some(function (u) { return u.indexOf(id) >= 0; });
    if (!listed) return { ok: false, error: "That song is not in the music list." };
  }
  try {
    var f = DriveApp.getFileById(id);
    if (f.getSize() > 12 * 1024 * 1024) return { ok: false, error: "That MP3 is over 12 MB. Use a smaller file." };
    var blob = f.getBlob(), mime = String(blob.getContentType() || "audio/mpeg");
    if (mime.indexOf("audio") !== 0 && mime !== "application/octet-stream") return { ok: false, error: "That Drive file is not an audio file." };
    return { ok: true, mime: mime.indexOf("audio") === 0 ? mime : "audio/mpeg", b64: Utilities.base64Encode(blob.getBytes()) };
  } catch (e) { return { ok: false, error: "Could not open that Drive file. Check it is shared as Anyone with the link, and that authorizeDrive was run. " + e }; }
}

// Reads the Drive link(s) the admin pasted: a shared folder (every picture in it) and/or single photo links, one per line
function awardsLoad_(ss, user, text) {
  if (!user.isAdmin) return { ok: false, error: "admin_only" };
  var toks = String(text || "").split(/[\s,]+/).filter(function (t) { return /^https?:\/\//i.test(t); });
  if (!toks.length) return { ok: false, error: "Paste a Google Drive folder link or photo links first." };
  var out = [], seen = {}, notes = [];
  toks.forEach(function (t) {
    var fo = t.match(/drive\.google\.com\/(?:drive\/(?:u\/\d+\/)?)?folders\/([\w-]+)/);
    var fi = t.match(/drive\.google\.com\/file\/d\/([\w-]+)/) || t.match(/drive\.google\.com\/(?:open|uc|thumbnail)\?(?:[^#]*&)?id=([\w-]+)/) || t.match(/docs\.google\.com\/uc\?(?:[^#]*&)?id=([\w-]+)/);
    if (fo) {
      try {
        var files = DriveApp.getFolderById(fo[1]).getFiles(), arr = [];
        while (files.hasNext()) { var f = files.next(); if (String(f.getMimeType()).indexOf("image/") === 0) arr.push({ id: f.getId(), n: f.getName() }); }
        arr.sort(function (a, b) { return a.n.localeCompare(b.n, undefined, { numeric: true, sensitivity: "base" }); });
        arr.forEach(function (x) { if (!seen[x.id]) { seen[x.id] = 1; out.push({ u: "https://drive.google.com/file/d/" + x.id + "/view", n: x.n }); } });
        if (!arr.length) notes.push("The folder has no pictures.");
      } catch (e) { notes.push("Could not open the folder. Share it as \"Anyone with the link\" and make sure this script is allowed to use Google Drive (run authorizeDrive once in the Apps Script editor). " + e); }
    } else if (fi) {
      if (!seen[fi[1]]) { seen[fi[1]] = 1; var nm = ""; try { nm = DriveApp.getFileById(fi[1]).getName(); } catch (e2) {} out.push({ u: "https://drive.google.com/file/d/" + fi[1] + "/view", n: nm }); }
    } else if (/\.(jpe?g|png|webp|gif)(\?|$)/i.test(t)) {
      if (!seen[t]) { seen[t] = 1; out.push({ u: t, n: "" }); }
    } else notes.push("Skipped a link that is not a Drive folder or photo: " + t.slice(0, 60));
  });
  if (!out.length) return { ok: false, error: notes.join(" ") || "No photos found." };
  return { ok: true, photos: out.slice(0, AWARD_MAX), note: notes.join(" "), cut: out.length > AWARD_MAX };
}

function normGroupTiers_(g) {
  g = g || {};
  var out = {};
  ["unit", "team", "branch"].forEach(function (k) {
    var src = (g[k] && g[k].length) ? g[k] : DEFAULT_GROUP_TIERS[k];

      .filter(function (t) { return t.name && t.ac > 0; }).sort(function (a, b) { return a.ac - b.ac; });
    if (!out[k].length) out[k] = DEFAULT_GROUP_TIERS[k].map(function (t) { return { name: t.name, ac: t.ac }; });

// Every contest carries its own measure and levels (the rules change each quarter).
// Older saved data with one shared basis/levels is filled into each contest here.
// Poster links (a shared Drive link or any image link): one for advisors, one for managers
function posterLink_(v) { return String(v === undefined || v === null ? "" : v).trim().slice(0, 1000); }
function normMed_(m) {
  var gb = m.basis || DEFAULT_MEDALLION.basis, gt = m.tiers || DEFAULT_MEDALLION.tiers;
  m.breakdown = bd_(m.breakdown);
  m.groupTiers = normGroupTiers_(m.groupTiers);
  (m.contests || []).forEach(function (c) {
    var mt = String(c.metric || gb.metric || "ac").toLowerCase();
    c.metric = (mt === "nsc" || mt === "lives") ? mt : "ac";
    c.newOnly = c.newOnly === undefined ? gb.newOnly !== false : c.newOnly !== false;
    if (!c.tiers || !c.tiers.length) c.tiers = gt.map(function (t) { return { name: t.name, ac: t.ac }; });
    c.posterAdv = posterLink_(c.posterAdv); c.posterMgr = posterLink_(c.posterMgr);
  });
  return m;
}

function setMedallion_(ss, user, m) {
  if (!user.isAdmin) return { ok: false, error: "admin_only" };
  m = m || {};
  var contests = [];
  var mmdd = /^(0[1-9]|1[0-2])-(0[1-9]|[12]\d|3[01])$/;
  (m.contests || []).forEach(function (c) {
    var name = String((c && c.name) || "").replace(/\s+/g, " ").trim().toUpperCase();

    if (!name && !st && !en) return;
    if (!name) throw new Error("contest_name");
    if (!mmdd.test(st) || !mmdd.test(en)) throw new Error("contest_dates");
    var metric = String((c && c.metric) || "ac").toLowerCase();
    if (metric !== "ac" && metric !== "nsc" && metric !== "lives") metric = "ac";
    var tiers = [];
    ((c && c.tiers) || []).forEach(function (t) {
      var tn = String((t && t.name) || "").replace(/\s+/g, " ").trim().toUpperCase();
      var ac = Number(t && t.ac);
      if (!tn && !ac) return;
      if (!tn || !isFinite(ac) || ac <= 0) throw new Error("tier_values");
      tiers.push({ name: tn, ac: Math.round(ac * 100) / 100 });
    });
    if (!tiers.length) throw new Error("tier_values");
    tiers.sort(function (a, b) { return a.ac - b.ac; });
    contests.push({ name: name, start: st, end: en, metric: metric, newOnly: !(c && c.newOnly === false), tiers: tiers, posterAdv: posterLink_(c && c.posterAdv), posterMgr: posterLink_(c && c.posterMgr) });
  });
  if (!contests.length) return { ok: false, error: "Add at least one contest." };
  // Manager levels (unit, sales team, branch): keep the saved ones when an older page does not send them
  var gt = m.groupTiers;
  if (!gt) gt = medallion_(ss).groupTiers;
  else {

    ["unit", "team", "branch"].forEach(function (k) {

      });

      chk[k] = arr;
    });

  var rows = sh.getLastRow() < 2 ? [] : sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues();

  // 1) Verify every table before touching any final tab
  var plan = tables.map(function (t) {
    var got = {};
    mine.forEach(function (r) { if (r[1] === t.name) got[Number(r[2])] = Number(r[3]); });
    var chunks = Math.max(1, Math.ceil(t.rows / body.chunkRows));
    var total = 0;
    for (var i = 0; i < chunks; i++) {
      if (got[i] === undefined) throw new Error("Missing chunk " + i + " for " + t.name + ". Retry the upload.");
      total += got[i];
    }
    if (total !== t.rows) throw new Error("Row count mismatch for " + t.name + ": expected " + t.rows + ", received " + total + ". Retry the upload.");
    var stage = getOrCreateTab_(ss, stageName_(t.tab));
    var values = [];
    if (t.rows > 0) {
      var cols = stage.getLastColumn();
      // An empty staging tab has no columns; say so clearly instead of failing inside Sheets
      if (cols < 1) throw new Error("Staging data for " + t.name + " is empty. Upload the file again.");
      values = stage.getRange(1, 1, t.rows, cols).getValues();
    }
    return { spec: t, values: values };
  });

  // 2) Apply each table to its final tab
  var added = {};
  plan.forEach(function (p) {
    var t = p.spec;
    var sh = getOrCreateTab_(ss, t.tab);
    var header = t.header || null;
    var rows = p.values;
    if (t.mode === "source") {
      // Replace only this source's rows (Source is column 1), keep the other source
      var old = readBody_(sh, header);
      var keep = old.filter(function (r) { return r[0] !== body.slot && r[0] !== ""; });
      rows = keep.concat(p.values);
    } else if (t.mode === "append") {
      // Daily update: add only the lines that are not in the tab yet (same advisor, date and amounts). Uploading the same day again, or a longer file, never doubles anything.
      var have = readBody_(sh, header), keyCols = t.name === "Settled" ? [1, 4, 5, 6, 7] : t.name === "Submitted" ? [1, 4, 5, 7, 8] : [1, 4, 5, 6, 7, 8, 9], seenK = {}, ssx = ss;
      var kf = function (r) { return keyCols.map(function (i) { var v = r[i]; if (v instanceof Date) return cell_(ssx, v, true); var n = Number(v); return v !== "" && v !== null && !isNaN(n) ? n.toFixed(2) : String(v).trim(); }).join("|"); };
      var firstAt = {};
      have.forEach(function (r, hi) { var k = kf(r); seenK[k] = (seenK[k] || 0) + 1; (firstAt[k] = firstAt[k] || []).push(hi); });
      // A line already stored without a policy number (uploaded before that column existed) is the same line: it is not added again, it just receives the policy number
      var fresh = p.values.filter(function (r) { var k = kf(r); if (seenK[k] > 0) { seenK[k]--; var hi = firstAt[k].shift(); if (t.name === "Settled" && r[9] && hi !== undefined && String(have[hi][9] || "") === "") { while (have[hi].length < 11) have[hi].push(""); have[hi][9] = r[9]; have[hi][10] = r[10] || ""; } if (t.name === "FY_Commission" && r[11] && hi !== undefined && String(have[hi][11] || "") === "") { while (have[hi].length < 12) have[hi].push(""); have[hi][11] = r[11]; } return false; } return true; });
      added[t.name] = { added: fresh.length, already: p.values.length - fresh.length };
      rows = have.concat(fresh);
    } else if (t.mode === "upsert") {
      // Match by first column (Code): update existing, add new
      var existing = readBody_(sh, header);
      var index = {};
      existing.forEach(function (r, i) { index[String(r[0])] = i; });
      rows = existing.slice();
      p.values.forEach(function (r) {
        var k = String(r[0]);
        if (index[k] !== undefined) rows[index[k]] = r;
        else { rows.push(r); index[k] = rows.length - 1; }
      });
    }
    writeTab_(sh, header, rows);
  });

  // 3) Clean up staging and this job's receipts
  plan.forEach(function (p) { getOrCreateTab_(ss, stageName_(p.spec.tab)).clearContents(); });
  var keepRec = all.filter(function (r) { return String(r[0]) !== String(body.jobId); });
  receipts.clearContents();
  if (keepRec.length) receipts.getRange(1, 1, keepRec.length, 4).setValues(keepRec);

  var summary = tables.map(function (t) { return t.name + " " + t.rows + " rows"; }).join(", ");
  appendLog_(ss, body, "complete", summary + (Object.keys(added).length ? " | added " + Object.keys(added).map(function (k) { return k + " " + added[k].added + " (already there " + added[k].already + ")"; }).join(", ") : ""));
  return { ok: true, added: added };
}

// ---------- Helpers ----------
function readBody_(sh, header) {
  var last = sh.getLastRow();
  var start = header ? 2 : 1;
  var cols = sh.getLastColumn();
  if (last === 0 || last < start || cols < 1) return [];
  return sh.getRange(start, 1, last - start + 1, cols).getValues();
}

function writeTab_(sh, header, rows) {
  var width = header ? header.length : (rows.length ? rows[0].length : 0);
  sh.clearContents();
  if (header && width) {
    sh.getRange(1, 1, 1, width).setValues([header]).setFontWeight("bold");
    sh.setFrozenRows(1);
  }
  if (rows.length && width) {
    var padded = rows.map(function (r) {
      var o = r.slice(0, width);
      while (o.length < width) o.push("");
      return o;
    });
    sh.getRange(header ? 2 : 1, 1, padded.length, width).setValues(padded);
  }
}

function appendLog_(ss, body, status, details) {
  var sh = getOrCreateTab_(ss, LOG_TAB);
  if (sh.getLastRow() === 0) sh.getRange(1, 1, 1, LOG_HEADER.length).setValues([LOG_HEADER]).setFontWeight("bold");
  sh.appendRow([new Date(), body.slot || "", body.file || "", body.by || "", status, details || ""]);
}

// ---------- One-time tab renamer ----------
// Run "renamePersistencyTabs" once from the Apps Script editor (select it in the function list, press Run, then read the Execution log).
// P13_OCTOBER 2026_Manager Persistency Summary -> 13th_October 2026_Projected   (or _Actual, read from the "Type" row inside the tab)
// P25_... -> 25th_October 2026_Actual / _Projected. Staging copies keep the STG_ in front.
// Any other sheet of the same file keeps its name after the period: 13th_October 2026_Actual_<sheet>.
// Older tabs without a period (P13_Manager Persistency Summary) are left as they are, because the sheet does not say which month they belong to.
function renamePersistencyTabs() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var taken = {};
  ss.getSheets().forEach(function (s) { taken[s.getName()] = true; });
  var done = [], skipped = [];
  function kindOf(sh) {
    var rows = Math.min(sh.getLastRow(), 30), cols = Math.min(sh.getLastColumn(), 2);
    if (rows < 1 || cols < 2) return "";
    var v = sh.getRange(1, 1, rows, cols).getValues();
    for (var i = 0; i < v.length; i++) {
      if (String(v[i][0]).trim() === "Type") {
        var t = String(v[i][1]).trim();
        if (/^projected/i.test(t)) return "Projected";
        if (/^actual/i.test(t)) return "Actual";
      }
    }
    return "";
  }
  ss.getSheets().forEach(function (sh) {
    var n = sh.getName(), stg = "", slotNo, per, rest = "", main;
    var m = n.match(/^(STG_)?P(13|25)_([A-Za-z]+ \d{4})_(.*)$/);
    if (m) { stg = m[1] || ""; slotNo = m[2]; per = m[3]; rest = m[4]; main = /manager|pers/i.test(rest); }
    else {
      m = n.match(/^(STG_)?(13|25)th_([A-Za-z]+ \d{4})(?:_(Actual|Projected))?(?:_(.*))?$/i);
      if (!m) return;
      stg = m[1] || ""; slotNo = m[2]; per = m[3]; rest = m[5] || ""; main = !m[5];
    }
    if (/agreement/i.test(rest)) return;
    per = per.toLowerCase().replace(/\b[a-z]/g, function (c) { return c.toUpperCase(); });
    var kd = kindOf(sh);
    var base = stg + slotNo + "th_" + per + (kd ? "_" + kd : "");
    var target = main ? base : base + "_" + rest;
    if (target === n) return;
    if (taken[target]) { skipped.push(n + " -> " + target + " (a tab with that name already exists)"); return; }
    sh.setName(target);
    delete taken[n]; taken[target] = true;
    done.push(n + " -> " + target + (kd ? "" : "  (no Type row found, so no Actual/Projected added)"));
  });
  bumpVer_();
  Logger.log("Renamed " + done.length + " tab(s):\n" + done.join("\n") + (skipped.length ? "\n\nSkipped:\n" + skipped.join("\n") : ""));
  return { renamed: done, skipped: skipped };
}

// The browser cannot draw a Google Drive picture onto a downloadable poster (Drive does not allow it), so the page asks for the picture here.
// Only photo links that are in REGISTERED USERS (or the branch logo) are served, so this cannot be used to fetch anything else.
function photo_(ss, url) {
  function no(why) { return { ok: false, error: "no_photo", detail: String(why || "") }; }
  try {
    url = String(url || "").trim();
    if (!url) return no("no link");
    var bgUrl = String(getSetting_(ss, "BIRTHDAY BACKGROUND") || "").trim(), isBg = !!bgUrl && url === bgUrl;
    var allowed = isBg || url === String(getSetting_(ss, "BRANCH LOGO") || "").trim() || users_(ss).some(function (r) { return String(r[10] || "").trim() === url; });
    if (!allowed) return no("link not found in REGISTERED USERS");
    var m = url.match(/drive\.google\.com\/file\/d\/([\w-]+)/) || url.match(/drive\.google\.com\/(?:open|uc|thumbnail)\?(?:[^#]*&)?id=([\w-]+)/) || url.match(/docs\.google\.com\/uc\?(?:[^#]*&)?id=([\w-]+)/);
    var key = (isBg ? "PB1:" : "P1:") + Utilities.base64EncodeWebSafe(m ? m[1] : url).slice(0, 80);
    var hit = cacheGet_(key);
    if (hit && hit.d) return { ok: true, dataUrl: hit.d };
    var blob = null, why = "";
    if (m) {
      try {
        var res = UrlFetchApp.fetch("https://drive.google.com/thumbnail?id=" + m[1] + "&sz=w" + (isBg ? 1000 : 400), { muteHttpExceptions: true, followRedirects: true });
        var ct = String(res.getHeaders()["Content-Type"] || res.getHeaders()["content-type"] || "");
        if (res.getResponseCode() === 200 && ct.indexOf("image") === 0) blob = res.getBlob(); else why = "fetch " + res.getResponseCode() + " " + ct;
      } catch (e1) { why = "fetch: " + e1; }
    } else if (/^https?:\/\//i.test(url)) {
      var r2 = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
      if (r2.getResponseCode() === 200) blob = r2.getBlob(); else why = "fetch " + r2.getResponseCode();
    }
    if (!blob) return no(why || "no picture returned");
    var bytes = blob.getBytes();
    if (bytes.length > (isBg ? 3000000 : 1500000)) return no("picture too large");
    var d = "data:" + (blob.getContentType() || "image/jpeg") + ";base64," + Utilities.base64Encode(bytes);
    cachePut_(key, { d: d }, 21600);
    return { ok: true, dataUrl: d };
  } catch (e) { return no(e); }
}
